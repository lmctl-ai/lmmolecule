cmake_minimum_required(VERSION 3.24)
project(lmmolecule VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMMOLECULE_FULL_STATIC "Link glibc statically too (LGPL relink duties; off per policy: glibc stays dynamic)" OFF)

# Embed schema and notices.
function(embed_file var file name)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${file}")
  file(READ "${file}" hex HEX)
  string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," hex "${hex}")
  set(${var} "inline constexpr unsigned char ${name}[] = {${hex}0}; inline constexpr std::size_t ${name}_size = sizeof(${name}) - 1;\n" PARENT_SCOPE)
endfunction()
set(GEN "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GEN}")
embed_file(SCHEMA_SRC "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmmolecule.dsl.json" embedded_schema)
file(WRITE "${GEN}/schema_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmmolecule { ${SCHEMA_SRC} }\n")
embed_file(NOTICES_SRC "${CMAKE_CURRENT_SOURCE_DIR}/THIRD_PARTY_NOTICES" embedded_notices)
file(WRITE "${GEN}/notices_embedded.hpp" "#pragma once\n#include <cstddef>\nnamespace lmmolecule { ${NOTICES_SRC} }\n")

include(cmake/OpenMM.cmake)

add_executable(lmmolecule cpp/main.cpp cpp/memory_budget.cpp)
target_compile_definitions(lmmolecule PRIVATE LMMOLECULE_VERSION="${PROJECT_VERSION}")
target_include_directories(lmmolecule PRIVATE cpp "${GEN}")
target_link_libraries(lmmolecule PRIVATE openmm)
target_compile_options(lmmolecule PRIVATE -Wall -Wextra)
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmmolecule PRIVATE -static-libstdc++ -static-libgcc)
  if(MINGW)
    # Windows keeps the MinGW runtime fully static (Rule 14 concerns Linux/glibc only).
    target_link_options(lmmolecule PRIVATE -static -Wl,--whole-archive -lpthread -Wl,--no-whole-archive)
  elseif(LMMOLECULE_FULL_STATIC)
    target_link_options(lmmolecule PRIVATE -static)
    if(UNIX)
      target_link_options(lmmolecule PRIVATE -Wl,--whole-archive -lpthread -Wl,--no-whole-archive)
    endif()
  endif()
  if(CMAKE_BUILD_TYPE STREQUAL "Release")
    target_link_options(lmmolecule PRIVATE -s)
  endif()
endif()
